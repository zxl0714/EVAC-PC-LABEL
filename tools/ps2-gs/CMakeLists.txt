# A driver can advertise descriptorBuffer while Granite rejects its layout or
# capacity. All consumers must follow the selected strategy, not that raw bit.
# Upstream investigation: https://github.com/Themaister/Granite/pull/157
# Keep the source view immutable; fail for review if the pinned code changes.
foreach(unit IN ITEMS shader memory_allocator descriptor_set)
  set(original "${PS2_GS_SOURCE}/Granite/vulkan/${unit}.cpp")
  file(READ "${original}" unit_text)
  set(raw_check "device->get_device_features().descriptor_buffer_features.descriptorBuffer")
  string(REGEX MATCHALL "device->get_device_features\\(\\)\\.descriptor_buffer_features\\.descriptorBuffer" raw_checks "${unit_text}")
  list(LENGTH raw_checks raw_count)
  if(NOT raw_count EQUAL 1)
    message(FATAL_ERROR "Review Granite ${unit} descriptor fallback after updating the source pin")
  endif()
  string(REPLACE "${raw_check}" "device->get_device_features().supports_descriptor_buffer" unit_text "${unit_text}")
  set(patched "${CMAKE_CURRENT_BINARY_DIR}/collection-${unit}.cpp")
  file(CONFIGURE OUTPUT "${patched}" CONTENT "@unit_text@" @ONLY)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${original}")
  get_target_property(granite_sources granite-vulkan SOURCES)
  list(TRANSFORM granite_sources REPLACE "^(.*[/])?${unit}[.]cpp$" "${patched}")
  set_property(TARGET granite-vulkan PROPERTY SOURCES "${granite_sources}")
endforeach()
# clang-cl treats -Wall as MSVC /Wall (every warning), rather than Clang's
# normal -Wall group. Preserve the upstream options with explicit forwarding.
function(collection_gs_clang_options directory)
  get_property(targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
  foreach(target IN LISTS targets)
    get_target_property(options "${target}" COMPILE_OPTIONS)
    if(options)
      set(forwarded)
      foreach(option IN LISTS options)
        if(option MATCHES "^-W" OR option STREQUAL "-msse3")
          list(APPEND forwarded "/clang:${option}")
        else()
          list(APPEND forwarded "${option}")
        endif()
      endforeach()
      set_property(TARGET "${target}" PROPERTY COMPILE_OPTIONS "${forwarded}")
    endif()
  endforeach()
  get_property(children DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
  foreach(child IN LISTS children)
    collection_gs_clang_options("${child}")
  endforeach()
endfunction()
